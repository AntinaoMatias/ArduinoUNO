#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <SPI.h>
#include <MFRC522.h>
#include <Keypad.h>
#include <Servo.h>

// --- CONFIGURACIÓN LCD I2C ---
LiquidCrystal_I2C lcd(0x27, 16, 2);

// --- CONFIGURACIÓN RFID ---
#define SDA_PIN 10
#define RST_PIN 9
MFRC522 mfrc522(SDA_PIN, RST_PIN);

// Lista de tarjetas autorizadas
String tarjetasAutorizadas[] = {
  "1E 48 59 06",
  "96 F8 6B 06"
};
const int totalTarjetas = 2;

// --- CONFIGURACIÓN TECLADO 4x4 (Invertido para corregir efecto espejo) ---
const byte FILAS = 4;
const byte COLUMNAS = 4;
char teclas[FILAS][COLUMNAS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte pinesFilas[FILAS] = {A0, 8, 7, 6};     // R4, R3, R2, R1
byte pinesColumnas[COLUMNAS] = {5, 4, 3, 2}; // C1, C2, C3, C4
Keypad teclado = Keypad(makeKeymap(teclas), pinesFilas, pinesColumnas, FILAS, COLUMNAS);

// --- SERVO, BOTÓN Y LED/BUZZER COMPARTIDO ---
Servo pestillo;
const int PIN_SERVO = A1;
const int PIN_BOTON_EMERGENCIA = A2; // Botón físico de 4 pines
const int LED_Y_BUZZER = A3;         // LED Verde y Buzzer comparten el pin A3 (apagado en reposo)

// --- CREDENCIALES Y SEGURIDAD ---
const String PIN_CORRECTO = "12345"; 
const String TOKEN_BLUETOOTH = "PASSWORD";
String pinIngresado = "";
int intentosFallidos = 0;

void setup() {
  // Inicializa el Puerto Serie por Hardware (Pines 0 y 1 para el HC-05)
  Serial.begin(9600);
  
  SPI.begin();
  mfrc522.PCD_Init();
  
  pinMode(LED_Y_BUZZER, OUTPUT);
  pinMode(PIN_BOTON_EMERGENCIA, INPUT_PULLUP); // Activa resistencia interna
  
  pestillo.attach(PIN_SERVO);
  pestillo.write(0); // Posición CERRADO (0 grados)

  lcd.init();                      
  lcd.backlight();
  
  digitalWrite(LED_Y_BUZZER, LOW); // LED apagado en reposo
  mensajeInicial();
}

void loop() {
  // 1. Revisar Botón de Emergencia Físico
  if (digitalRead(PIN_BOTON_EMERGENCIA) == LOW) {
    delay(30);
    if (digitalRead(PIN_BOTON_EMERGENCIA) == LOW) {
      concederAcceso("Boton Emergencia");
      delay(1000); // Pausa para evitar bucle si se mantiene presionado
    }
  }

  // 2. Bloqueo por fuerza bruta (50 intentos fallidos)
  if (intentosFallidos >= 50) {
    lcd.clear();
    lcd.print("SISTEMA BLOQUEADO");
    lcd.setCursor(0, 1);
    lcd.print("Espere 30 seg...");
    
    // Alerta visual y sonora durante el bloqueo
    for(int i=0; i<30; i++) {
      digitalWrite(LED_Y_BUZZER, HIGH);
      delay(500);
      digitalWrite(LED_Y_BUZZER, LOW);
      delay(500);
    }
    
    intentosFallidos = 0;
    mensajeInicial();
  }

  // 3. Teclado Matricial
  char tecla = teclado.getKey();
  if (tecla) {
    // Destello y pitido corto al presionar tecla
    digitalWrite(LED_Y_BUZZER, HIGH); delay(30); digitalWrite(LED_Y_BUZZER, LOW);
    
    if (tecla == '*') { 
      pinIngresado = "";
      mensajeInicial();
    } 
    else if (tecla == '#') { 
      if (pinIngresado == PIN_CORRECTO) {
        concederAcceso("PIN Correcto");
      } else {
        denegarAcceso("PIN Incorrecto");
      }
      pinIngresado = "";
    } 
    else {
      pinIngresado += tecla;
      lcd.setCursor(pinIngresado.length() - 1, 1);
      lcd.print("*"); 
    }
  }

  // 4. Bluetooth (HC-05 conectado a los pines 0 [RX] y 1 [TX])
  if (Serial.available() > 0) {
    String comandoBT = Serial.readStringUntil('\n');
    comandoBT.trim();
    if (comandoBT == TOKEN_BLUETOOTH) {
      concederAcceso("Acceso Bluetooth");
    } else if (comandoBT.length() > 0) {
      denegarAcceso("Token Invalido");
    }
  }

  // 5. Tarjeta RFID
  if (mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {
    String uidLeido = "";
    for (byte i = 0; i < mfrc522.uid.size; i++) {
      uidLeido += String(mfrc522.uid.uidByte[i] < 0x10 ? " 0" : " ");
      uidLeido += String(mfrc522.uid.uidByte[i], HEX);
    }
    uidLeido.trim();
    uidLeido.toUpperCase();

    // Validar si el UID está en la lista de autorizadas
    bool tarjetaValida = false;
    for (int i = 0; i < totalTarjetas; i++) {
      if (uidLeido == tarjetasAutorizadas[i]) {
        tarjetaValida = true;
        break;
      }
    }

    if (tarjetaValida) {
      concederAcceso("Tarjeta Valida");
    } else {
      denegarAcceso("Tarjeta Denegada");
    }
    mfrc522.PICC_HaltA();
  }
}

void mensajeInicial() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Acerque Tarjeta");
  lcd.setCursor(0, 1);
  lcd.print("o PIN + [#]:");
  digitalWrite(LED_Y_BUZZER, LOW); // LED apagado mientras espera en reposo
}

void concederAcceso(String metodo) {
  intentosFallidos = 0;
  lcd.clear();
  lcd.print("ACCESO PERMITIDO");
  lcd.setCursor(0, 1);
  lcd.print(metodo);
  
  // Parpadeo del LED/Buzzer indicando éxito
  for(int i=0; i<4; i++) {
    digitalWrite(LED_Y_BUZZER, HIGH); delay(100);
    digitalWrite(LED_Y_BUZZER, LOW); delay(100);
  }

  pestillo.write(90); // Gira a 90 grados (ABIERTO)
  delay(4000);        // Espera 4 segundos
  pestillo.write(0);  // Regresa a 0 grados (CERRADO)
  
  mensajeInicial();
}

void denegarAcceso(String motivo) {
  intentosFallidos++;
  lcd.clear();
  lcd.print("ACCESO DENEGADO");
  lcd.setCursor(0, 1);
  lcd.print(motivo);
  
  // Destello de error
  digitalWrite(LED_Y_BUZZER, HIGH);
  delay(1000);
  digitalWrite(LED_Y_BUZZER, LOW);
  delay(1000);
  
  mensajeInicial();
}
